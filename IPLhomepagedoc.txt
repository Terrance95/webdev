        <script>
            var bookButtons = document.querySelectorAll('.book-btn');
            for (var i = 0; i < bookButtons.length; i++) {
                bookButtons[i].addEventListener('click', function(event) {
                    var matchTeams = event.target.getAttribute('data-teams');
                    var matchDate = event.target.getAttribute('data-date');
                    var matchVenue = event.target.getAttribute('data-venue');

                    var targetUrl = 'Booking.html' +
                        '?teams=' + encodeURIComponent(matchTeams) +
                        '&date=' + encodeURIComponent(matchDate) +
                        '&venue=' + encodeURIComponent(matchVenue);

                    window.open(targetUrl, '_blank', 'noopener,noreferrer');
                });
            }
        </script>
